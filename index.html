Потрібне налаштування Supabase 🛠️

Схоже, таблиця cloud_saves ще не створена у вашому проекті Supabase.
Щоб хмарна синхронізація запрацювала, перейдіть до Supabase SQL Editor та виконайте наступний код: 

CREATE TABLE cloud_saves (
  user_id uuid REFERENCES auth.users NOT NULL PRIMARY KEY,
  email text,
  role text DEFAULT 'patient',
  shared_with text[] DEFAULT '{}',
  history_json jsonb DEFAULT '[]',
  colors_json jsonb DEFAULT '{}'
);

-- Базові політики доступу (RLS)
ALTER TABLE cloud_saves ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can manage their own data" ON cloud_saves
  FOR ALL USING (auth.uid() = user_id);

CREATE POLICY "Doctors can read shared data" ON cloud_saves
  FOR SELECT USING (auth.jwt()->>'email' = ANY(shared_with));
